
from flask import Flask

app = Flask(__name__)

@app.route('/')
def home():
    return "Hello World! I am inside Kubernetes and Docker."

if __name__ == '__main__':
    app.run(host='0.0.0.0', port=5000)



Flask==3.1.2



FROM python:3.12-slim
WORKDIR /app
COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt
COPY app.py .
EXPOSE 5000
CMD ["python", "app.py"]



apiVersion: apps/v1
kind: Deployment
metadata:
  name: student-docker-deployment
spec:
  replicas: 2
  selector:
    matchLabels:
      app: student-docker-app
  template:
    metadata:
      labels:
        app: student-docker-app
    spec:
      containers:
        - name: student-docker-container
          image: student-docker-app:v1
          imagePullPolicy: Never
          ports:
            - containerPort: 5000



apiVersion: v1
kind: Service
metadata:
  name: student-docker-service
spec:
  type: NodePort
  selector:
    app: student-docker-app
  ports:
    - port: 5000
      targetPort: 5000
      nodePort: 30080


docker --version
kubectl get nodes


docker build -t student-docker-app:v1 .

kubectl apply -f deployment.yaml

kubectl get pods
kubectl get deployments
kubectl get services
